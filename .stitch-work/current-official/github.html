import { definePulseBot, type PullRequestContext } from '@gitpulse/bot-sdk';

export default definePulseBot({
  name: 'smart-reviewer-v2',
  async onPullRequestOpen(ctx: PullRequestContext) {
    // 自動偵測代碼異動幅度並推薦指定架構師
    const diffSize = await ctx.getDiffMetrics();
    if (diffSize.insertions > 500) {
      await ctx.addLabels(['needs-arch-review']);
      await ctx.assignReviewers(['tech-lead-alpha']);
    }
    await ctx.comment('🚀 GitPulse 已建立沙盒環境：預覽網址已生成');
  }
});
