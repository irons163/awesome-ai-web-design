import { Agent, Tool, MemoryStore } from '@voltagent/core';
import { PostgresMemory } from '@voltagent/memory-pg';

// Initialize deterministic state memory
const memory = new PostgresMemory({
  connectionString: process.env.DATABASE_URL,
  namespace: 'enterprise-ops',
});

// Define typed autonomous agent
export const leadAnalyst = new Agent({
  name: 'Lead Market Analyst',
  model: 'claude-3-5-sonnet-20241022',
  memory,
  tools: [
    new Tool({
      name: 'queryVectorDb',
      description: 'Execute hybrid sparse/dense vector search',
      parameters: z.object({ query: z.string(), topK: z.number().default(5) }),
      handler: async ({ query, topK }) => {
        return await memory.retrieveVectors(query, topK);
      },
    }),
  ],
  guardrails: { maxTokens: 4096, hallucinationCheck: true }
});

supervisor-mesh.ts Strict Multi-Agent Routing 
import { Supervisor, createSpecialist } from '@voltagent/supervisor';

// Instantiate domain specialists
const researchSpecialist = createSpecialist({
  id: 'research',
  model: 'claude-3-5-sonnet',
  capabilities: ['web_search', 'arxiv_lookup']
});

const coderSpecialist = createSpecialist({
  id: 'synthesis',
  model: 'gpt-4o',
  capabilities: ['ast_generation', 'test_runner']
});

// Mesh orchestrator with state consensus
export const engineeringSupervisor = new Supervisor({
  name: 'Engineering Directorate',
  agents: [researchSpecialist, coderSpecialist],
  routingStrategy: 'dynamic_consensus',
  sharedMemory: true,
  maxIterations: 10
});

1. Centralized Coordination

A single authoritative supervisor ensures task decomposition, deduplication, and non-cyclical execution graphs.

2. Specialized Agent Roles

Every sub-agent is locked to specific system prompts, token budgets, and tailored toolsets to eliminate drift.

3. Shared Memory System

Unified vector and key-value state ensures contextual knowledge passes between agents without context-window stuffing.

4. Dynamic Agent Selection

The supervisor decides which agent to invoke next based on intermediate validation feedback loops.
