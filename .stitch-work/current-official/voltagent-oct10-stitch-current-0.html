Runtime Engine: Node.js 22 / Bun / Deno 
import { Agent, tool } from "@voltagent/core";
import { z } from "zod";

// 1. Strongly typed tool with runtime Zod guard
export const executeRefundTool = tool({
  name: "processRefund",
  description: "Initiate transactional refund through Stripe gateway",
  schema: z.object({
    chargeId: z.string().startsWith("ch_"),
    amountCents: z.number().int().positive(),
    reason: z.enum(["fraud", "customer_request"]),
  }),
  execute: async ({ chargeId, amountCents, reason }) => {
    const refund = await stripe.refunds.create({ charge: chargeId, amount: amountCents });
    return { status: "success", refundId: refund.id };
  },
});

// 2. Bootstrap enterprise supervisor instance
export const customerSupportAgent = new Agent({
  id: "enterprise-support-lead",
  instructions: "You are the senior billing supervisor. Validate policies before tools.",
  tools: [executeRefundTool],
  maxSteps: 5,
  memoryProvider: "redis://cache.internal.net:6379",
});

Declarative Directed Acyclic Graph 
import { Workflow } from "@voltagent/workflow";

export const analyzeCompany = new Workflow({ name: "FinAudit" })
  .step("fetchTicker", async (ctx) => fetchSymbol(ctx.input.symbol))
  // Parallel Fan-Out Execution
  .andAll([
    (ctx) => scrapeFilings(ctx.fetchTicker),
    (ctx) => computeRatios(ctx.fetchTicker)
  ])
  .step("compileReport", async (ctx) => generatePdfReport(ctx))
  // Pause for human-in-the-loop signoff
  .checkpoint({ storage: "postgres", pauseOnApproval: true });
