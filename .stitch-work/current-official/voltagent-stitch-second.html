WORKFLOW CHAIN API

Orchestrate your agents

Build complex agent workflows with a simple, declarative API. Combine parallel execution, transactional checkpoints, and strict validation.

TS workflow.ts
Zod Ready 
import { createWorkflowChain, agent } from '@voltagent/core';
import { z } from 'zod';

// Define specialized autonomous agent workers
const researcher = agent({ id: 'market-researcher', model: 'claude-3-5-sonnet' });
const analyst = agent({ id: 'financial-analyst', model: 'gpt-4o' });
const reporter = agent({ id: 'doc-generator', model: 'claude-3-5-haiku' });

// Declarative DAG with automatic schema inference & fan-out
export const diligenceWorkflow = createWorkflowChain('due-diligence')
  .input(z.object({ companyTicker: z.string(), depth: z.enum(['standard', 'deep']) }))
  .andAll([
    researcher.step('scrapeFilings'),
    analyst.step('computeRatios')
  ])
  .then(reporter.step('compileReport'))
  .checkpoint({ storage: 'postgres', pauseOnApproval: true });
